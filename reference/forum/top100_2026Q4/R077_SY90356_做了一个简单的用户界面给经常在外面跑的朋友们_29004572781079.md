def background_task1(label):

first_order = first_order_factory(pc_fields, ts_ops)

init_decay = 6

for alpha in first_order:

def send_notifications(s,body):

# 邮件发送者和接收者163举例

sender = '发送邮箱@163.com'

receiver = '接收邮箱@163.com'

# 创建一个带有附件的实例

message = MIMEMultipart()

message['From'] = sender

message['To'] = receiver

message['Subject'] = f"BRAIN SIMULATION DONE({body})"

response = s.post(' [https://api.worldquantbrain.com/authentication').content](https://api.worldquantbrain.com/authentication').content)

decoded_data = response.decode('utf-8')

data_dict = json.loads(decoded_data)

user_id = data_dict['user']['id']

full_body = f"{user_id}{body}"

# 邮件正文内容

message.attach(MIMEText(full_body, 'plain', 'utf-8'))

# 发送邮件服务器163举例

smtp_server = 'smtp.163.com'

smtp_port = 465  # SSL端口

# 发送者邮箱的用户名和密码

username = '发送邮箱的用户名'

password = '这个是发送邮箱的验证字符串'

try:

smtp = smtplib.SMTP_SSL(smtp_server, smtp_port)  # 连接到服务器

smtp.login(username, password)  # 登录验证

smtp.sendmail(message['From'], message['To'], message.as_string())  # 发送邮件

print("邮件发送成功")

except smtplib.SMTPException as e:

print("邮件发送失败", e)

finally:

smtp.quit()

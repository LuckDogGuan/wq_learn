概述：

代码可以支持：

在指定地区一键按照powerpoolselected等权配置分数（适合powerpool combine高的）

在指定地区按照因子命名进行等权配置（默认名字为SELECT）

另：实操中存在fastd1等无法赋分的因子，默认跳过导致分数不够十万，需要用osmosis scale配平

# -*- coding: utf-8 -*-

"""

等权赋分工具 (PowerPoolSelected)

================================

从 cell 6(alpha_allocator) 抽出的独立脚本, 增加前置清理步骤。

流程(对每个配置的地区):

1. 拉取该地区 REGULAR + delay=1 的全部alpha;

2. 把"当前有分数但没带 PowerPoolSelected tag"的因子分数清零(最多尝试3次);

3. 只对真正带 PowerPoolSelected tag 的因子等权赋分(每因子最多尝试3次, 仍失败则跳过, 不回头调整)。

用法:

1. 在下方 power_pool_regions 列表里加入要处理的地区代码即可;

2. DRY_RUN = True  -> 只打印方案, 不实际PATCH(含清零都不执行)

DRY_RUN = False -> 真实执行

"""

import urllib.parse

import logging

import time

import pandas as pd

from datetime import datetime

from machine_lib import login

logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")

logger = logging.getLogger("alpha_allocator")

# dry-run 开关: True=只出分配方案、不实际PATCH赋分; False=真实赋分

DRY_RUN = True

def fetch_region_alphas(s, region, start_date, mode='raw', limit=100, offset=0):

"""

分页获取指定地区的alpha(含当前 osmosisPoints)。

mode: 'raw'   -> REGULAR + delay=1, 不带tag过滤(服务端不支持tags过滤)

'select'-> type!=SUPER 且 name=SELECT

"""

alphas_data = []

start_date_str = urllib.parse.quote(start_date.astimezone().isoformat(timespec='seconds'))

while True:

if mode == 'select':

url = (

f" [https://api.worldquantbrain.com/users/self/alphas](https://api.worldquantbrain.com/users/self/alphas) ?"

f"limit={limit}&offset={offset}"

f"&dateCreated%3E={start_date_str}"

f"&settings.region={region}"

f"&settings.delay=1"

f"&status!=UNSUBMITTED%1FIS-FAIL"

f"&hidden=false"

f"&order=-dateSubmitted"

f"&type!=SUPER"

f"&name=SELECT"

)

else:  # raw

url = (

f" [https://api.worldquantbrain.com/users/self/alphas](https://api.worldquantbrain.com/users/self/alphas) ?"

f"limit={limit}&offset={offset}"

f"&dateCreated%3E={start_date_str}"

f"&settings.region={region}"

f"&settings.delay=1"

f"&status!=UNSUBMITTED%1FIS-FAIL"

f"&hidden=false"

f"&order=-dateSubmitted"

f"&type=REGULAR"

)

try:

resp = s.get(url)

if resp.status_code != 200:

logger.error(f"[{region}] 请求出错，状态码：{resp.status_code}")

break

data = resp.json()

results = data.get("results", [])

alphas_data.extend(results)

if offset + len(results) >= data.get("count", 0) or len(results) < limit:

break

offset += limit

except Exception as e:

logger.error(f"[{region}] 数据获取异常：{e}")

break

alpha_metrics = []

for item in alphas_data:

is_data = item.get('is', {})

alpha_metrics.append({

'id': item['id'],

'region': region,

'mode': mode,

'current_osmosis': int(item.get('osmosisPoints') or 0),

'fitness': is_data.get('fitness', 0.0),

'longCount': is_data.get('longCount', 0.0),

'shortCount': is_data.get('shortCount', 0.0),

'turnover': is_data.get('turnover', 0.0),

'returns': is_data.get('returns', 0.0),

'drawdown': is_data.get('drawdown', 0.0),

'margin': is_data.get('margin', 0.0),

'sharpe': is_data.get('sharpe', 0.0),

'name': item.get('name', ''),

'type': item.get('type', ''),

'tags': item.get('tags', [])

})

logger.info(f"[{region}][{mode}] 成功获取 {len(alpha_metrics)} 个alpha")

return alpha_metrics

def clear_scores_of_untagged(session, items, region):

"""

把"有分数(>0)但没带 PowerPoolSelected tag"的因子分数清零。

每个因子最多尝试3次; 失败则记录, 交由人工处理。

DRY_RUN=True 时只打印方案, 不实际调用PATCH。

"""

total = len(items)

result = {'total': total, 'planned': 0, 'cleared': 0, 'fail': 0, 'failed_ids': []}

if total == 0:

logger.info(f"[{region}] 没有需要清零的[有分但无tag]因子")

return result

logger.info(f"\n[{region}] 开始清理 {total} 个[有分数但无 PowerPoolSelected tag]的因子...")

for item in items:

alpha_id = item['id']

cur_score = int(item.get('current_osmosis') or 0)

if DRY_RUN:

result['planned'] += 1

logger.info(f"○(dry-run) 应清零 {alpha_id} (当前{cur_score}分, 无tag)")

continue

done = False

last_code = None

for attempt in range(1, 4):

# 依次尝试 置0 / 置null

for payload in ({"osmosisPoints": 0}, {"osmosisPoints": None}):

try:

resp = session.patch(

f" [https://api.worldquantbrain.com/alphas/{alpha_id}](https://api.worldquantbrain.com/alphas/{alpha_id}) ", json=payload

)

if resp.status_code == 200:

done = True

break

last_code = resp.status_code

except Exception as e:

last_code = f"异常:{str(e)}"

if done:

break

if attempt < 3:

time.sleep(0.5)

else:

last_code = last_code  # 保持原样

if done:

result['cleared'] += 1

logger.info(f"✓ 已清零 {alpha_id} (原{cur_score}分)")

else:

result['fail'] += 1

result['failed_ids'].append(alpha_id)

logger.warning(f"✗ {alpha_id} 清零失败(尝试3次): {last_code}，请人工处理")

if not DRY_RUN:

logger.info(f"[{region}] 清零完成: 成功 {result['cleared']} / {total}, 失败 {result['fail']}")

else:

logger.info(f"[{region}] (dry-run) 应清零 {result['planned']} / {total}")

return result

def allocate_equal_weights(n, total_points=100000):

"""等权分配分数，返回每个因子的目标分数列表"""

if n == 0:

return []

base_score = total_points // n

remainder = total_points % n

scores = [base_score] * n

for i in range(remainder):

scores[i] += 1

return scores

def update_alpha_scores(session, df, total_points=100000):

"""

更新alpha分数：按等权方案，每个因子最多尝试3次；

仍失败的因子直接跳过(由用户后续人工处理)，不回头重新调整成功因子的分数

"""

total_count = len(df)

if total_count == 0:

return df, {'total': 0, 'success': 0, 'fail': 0, 'failed_ids': []}

df['allocated_score'] = 0

df['update_status'] = '待更新'

# 一次性按总数等权切分(失败因子放弃自己那份, 不分配给别人)

scores = allocate_equal_weights(total_count, total_points)

MAX_ATTEMPTS = 3      # 每个因子最多尝试次数

ATTEMPT_PAUSE = 0.5   # 失败后的重试间隔(秒)

success_count = 0

fail_count = 0

failed_ids = []

for i, idx in enumerate(df.index):

alpha_id = df.at[idx, 'id']

new_score = scores[i]

if DRY_RUN:

df.at[idx, 'allocated_score'] = new_score

df.at[idx, 'update_status'] = '成功'

success_count += 1

logger.info(f"○(dry-run) {alpha_id} 方案分配 {new_score} 分")

continue

last_code = None

ok = False

for attempt in range(1, MAX_ATTEMPTS + 1):

try:

update_url = f" [https://api.worldquantbrain.com/alphas/{alpha_id}](https://api.worldquantbrain.com/alphas/{alpha_id}) "

response = session.patch(update_url, json={"osmosisPoints": int(new_score)})

if response.status_code == 200:

ok = True

break

last_code = response.status_code

if attempt < MAX_ATTEMPTS:

logger.warning(f"✗ {alpha_id} 第{attempt}次失败(码{last_code})，{ATTEMPT_PAUSE}s后重试...")

time.sleep(ATTEMPT_PAUSE)

except Exception as e:

last_code = f"异常:{str(e)}"

if attempt < MAX_ATTEMPTS:

logger.warning(f"✗ {alpha_id} 第{attempt}次异常: {str(e)}，{ATTEMPT_PAUSE}s后重试...")

time.sleep(ATTEMPT_PAUSE)

if ok:

df.at[idx, 'allocated_score'] = new_score

df.at[idx, 'update_status'] = '成功'

success_count += 1

logger.info(f"✓ {alpha_id} 分配 {new_score} 分")

else:

df.at[idx, 'update_status'] = f'失败(码{last_code})'

fail_count += 1

failed_ids.append(alpha_id)

logger.warning(f"✗ {alpha_id} 尝试{MAX_ATTEMPTS}次仍失败: {last_code}，跳过(请人工处理)")

# 仅打印校验, 不做任何回补调整

success_sum = int(df[df['update_status'] == '成功']['allocated_score'].sum())

logger.info(f"[校验] 成功因子实际合计: {success_sum} / 目标 {total_points} (失败因子分数已放弃, 不回头调整)")

return df, {

'total': total_count,

'success': success_count,

'fail': fail_count,

'failed_ids': failed_ids

}

def process_single_region(session, region, filter_mode, advisor_date, page_limit, page_offset, total_points):

"""处理单个地区：先清理无tag但有分的因子, 再对带tag因子等权赋分"""

logger.info(f"\n{'='*80}")

logger.info(f"开始处理地区: {region} | 模式: {filter_mode}")

logger.info(f"{'='*80}")

clear_result = None

if filter_mode == 'powerpool':

# 1) 拉全量 REGULAR+delay=1

raw = fetch_region_alphas(

s=session, region=region, start_date=advisor_date,

mode='raw', limit=page_limit, offset=page_offset

)

tagged = [x for x in raw if 'PowerPoolSelected' in (x.get('tags') or [])]

untagged_scored = [

x for x in raw

if 'PowerPoolSelected' not in (x.get('tags') or []) and x.get('current_osmosis', 0) > 0

]

logger.info(f"[{region}] 共{len(raw)}个: 带tag {len(tagged)}个 | 有分但无tag {len(untagged_scored)}个")

# 2) 先清零 [有分但无tag] 的因子

clear_result = clear_scores_of_untagged(session, untagged_scored, region)

# 3) 只对带tag因子等权赋分

if not tagged:

logger.warning(f"[{region}][powerpool] 没有带 PowerPoolSelected tag 的因子, 跳过赋分")

return None, None, clear_result

df = pd.DataFrame(tagged)

else:  # select

metrics = fetch_region_alphas(

s=session, region=region, start_date=advisor_date,

mode='select', limit=page_limit, offset=page_offset

)

if not metrics:

logger.warning(f"[{region}][select] 未获取到有效alpha数据")

return None, None, None

df = pd.DataFrame(metrics)

logger.info(f"[{region}][{filter_mode}] 待赋分因子数: {len(df)}")

df, update_results = update_alpha_scores(session, df, total_points=total_points)

# 预览

logger.info(f"\n[{region}][{filter_mode}] 分数分配方案预览（前10个）:")

preview_df = df.head(10)

for _, row in preview_df.iterrows():

logger.info(f"{row['id'][:12]} | 类型:{row['type']} | 分配:{int(row['allocated_score'])}分 | 状态:{row['update_status']}")

total_allocated = df['allocated_score'].sum()

success_mask = df['update_status'] == '成功'

logger.info(f"\n[{region}][{filter_mode}] 总分配分数: {total_allocated} (设定: {total_points})")

logger.info(f"[{region}][{filter_mode}] 成功因子数: {success_mask.sum()}/{len(df)}")

logger.info(f"[{region}][{filter_mode}] 失败因子数: {(~success_mask).sum()}")

return df, update_results, clear_result

def print_summary(all_results, region_configs, total_points=100000):

"""打印所有地区的汇总报告"""

logger.info(f"\n{'='*80}")

logger.info("全部分区处理完成 - 汇总报告")

logger.info(f"{'='*80}")

total_alphas = 0

total_success = 0

total_fail = 0

total_cleared = 0

total_clear_fail = 0

for region, filter_mode in region_configs:

key = f"{region}_{filter_mode}"

if key not in all_results:

continue

df, update_result, clear_result = all_results[key]

# 清理汇总

if clear_result is not None:

logger.info(f"\n[{region}] 无tag有分因子清理: 计划{clear_result['planned'] + clear_result['cleared']}, "

f"成功{clear_result['cleared']}, 失败{clear_result['fail']}")

total_cleared += clear_result['cleared']

total_clear_fail += clear_result['fail']

if df is None or update_result is None:

logger.info(f"\n[{region}][{filter_mode}] 无数据")

continue

count = len(df)

total_alphas += count

logger.info(f"\n[{region}][{filter_mode}] 处理数量: {count}")

logger.info(f"[{region}][{filter_mode}] 地区总分: {total_points}")

logger.info(f"[{region}][{filter_mode}] 更新成功: {update_result['success']}/{update_result['total']}")

logger.info(f"[{region}][{filter_mode}] 更新失败: {update_result['fail']}")

if update_result['failed_ids']:

logger.info(f"[{region}][{filter_mode}] 赋分失败ID: {', '.join(update_result['failed_ids'][:5])}")

if len(update_result['failed_ids']) > 5:

logger.info(f"[{region}][{filter_mode}] ... 等共 {len(update_result['failed_ids'])} 个")

total_success += update_result['success']

total_fail += update_result['fail']

logger.info(f"\n{'='*80}")

logger.info(f"总计赋分处理: {total_alphas} 个alpha")

logger.info(f"总计赋分成功: {total_success}")

logger.info(f"总计赋分失败: {total_fail}")

logger.info(f"总计清理成功(无tag有分): {total_cleared}")

logger.info(f"总计清理失败: {total_clear_fail}")

logger.info(f"{'='*80}")

if __name__ == '__main__':

# 配置参数

advisor_date = datetime(2020, 1, 1)

page_limit = 100

page_offset = 0

# 需要处理的地区列表: 要跑哪个地区就把区域代码加入下面的列表

# (流程: 先清零该地区[有分但无tag]的因子, 再对带 PowerPoolSelected tag 的因子等权赋分)

power_pool_regions = ['ASI','EUR','GLB','AMR','JPN']

# SELECT 名称地区列表 (name=SELECT, type!=SUPER)

select_regions = []  # 例如 'HKG', 'KOR'

# 每个地区总分

region_total_points = 100000

logger.info(f"{'='*80}")

logger.info("ALPHA因子批量等权赋分工具(先清理无tag残留, 再对带tag因子赋分)")

logger.info(f"{'='*80}")

logger.info(f"PowerPool模式地区: {', '.join(power_pool_regions)} (tag=PowerPoolSelected, type=REGULAR)")

logger.info(f"SELECT模式地区: {', '.join(select_regions)} (name=SELECT, type!=SUPER)")

logger.info(f"每个地区总分: {region_total_points}")

logger.info(f"分配方式: 等权 (总分/因子数)，失败尝试3次后跳过(不回补)")

logger.info(f"当前模式: {'DRY-RUN(只出方案,不实际赋分/清理)' if DRY_RUN else '真实执行'}")

logger.info(f"{'='*80}")

session = login()

logger.info("登录成功")

all_results = {}

region_configs = []

# 处理 PowerPool 地区

for region in power_pool_regions:

df, update_result, clear_result = process_single_region(

session, region, 'powerpool', advisor_date, page_limit, page_offset, region_total_points

)

key = f"{region}_powerpool"

all_results[key] = (df, update_result, clear_result)

region_configs.append((region, 'powerpool'))

# 处理 SELECT 地区

for region in select_regions:

df, update_result, clear_result = process_single_region(

session, region, 'select', advisor_date, page_limit, page_offset, region_total_points

)

key = f"{region}_select"

all_results[key] = (df, update_result, clear_result)

region_configs.append((region, 'select'))

print_summary(all_results, region_configs, region_total_points)

logger.info("程序执行完成")

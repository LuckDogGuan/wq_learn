## 完整代码  `client.py`

为了方便没办法登陆Gist的同学，这里也贴一下完整代码，请将本代码保存到  [`client.py`](http://client.py)  中，上篇代码保存到  `models.py`  中：

```
"""高级应用接口"""

import asyncio
import logging
from datetime import datetime, timedelta
from pathlib import Path
from typing import Any, Self, TypeVar
from zoneinfo import ZoneInfo

import httpx
from pydantic import BaseModel, ConfigDict, field_validator
from tortoise import Tortoise
from tortoise.exceptions import ConfigurationError
from tortoise.expressions import Q

from models import (
    ExternalModel,
    HTTPMethod,
    RegularAlpha,
    ResponseStatus,
    SuperAlpha,
    User,
)

type JSONLike = dict | list

T = TypeVar("T", bound=ExternalModel)

class NifflerResponse(BaseModel):
    model_config = ConfigDict(use_enum_values=True, arbitrary_types_allowed=True)

    status: ResponseStatus
    content: JSONLike
    raw: httpx.Response

    @field_validator("status", mode="before")
    @classmethod
    def validate_status(cls, v: str | ResponseStatus) -> ResponseStatus:
        if isinstance(v, str):
            return ResponseStatus(v)
        return v

class TokenExpiredError(Exception):
    pass

class BadRequestError(Exception):
    pass

class NifflerClient:
    """平台接口交互工具"""

    def __init__(
        self,
        connection: str | Path = "sqlite://:memory:",
        username: str = "username",
        password: str = "password",
    ) -> None:
        if isinstance(connection, Path):
            conn = f"sqlite:///{connection.absolute()}"
        else:
            conn = connection
        self._connection = conn
        self._logger = logging.getLogger(__name__)
        self._user: User | None = None
        self._cache: dict[str, Any] = {}

        self._username = username
        self._password = password
        self._expire_redun = timedelta(hours=0.5)

    async def send_request(
        self,
        client: httpx.AsyncClient,
        method: HTTPMethod,
        url: str,
        params: dict | None = None,
        payload: JSONLike | None = None,
        max_retry: int | None = None,
    ) -> NifflerResponse:
        """通用请求发送器

        NifflerClient 核心交互方法, 向平台发送请求, 支持 Get/Get-Later/Post/Patch, 实现了自动重试和超限休眠.
        返回结果抽象为 NifflerResponse, 既包含结果内容, 也包含原始返回内容.
        """

        if max_retry is None:
            max_retry = 10

        match HTTPMethod(method):
            case HTTPMethod.GET | HTTPMethod.GET_LATER:
                method_core: str = "GET"
            case _:
                method_core: str = method.value

        ntry = 0
        while max_retry - ntry > 0:
            try:
                res = await client.request(
                    method=method_core,
                    url=url,
                    timeout=None,
                    params=params,
                    json=payload,
                )
                if res.status_code in [401, 429]:
                    # Re-Send
                    self._user = await self.get_user()
                    client.cookies.set("t", self._user.cookie)
                    res = await client.request(
                        method=method_core,
                        url=url,
                        timeout=None,
                        params=params,
                        json=payload,
                    )
                match res.status_code:
                    case 401:  # 登陆失败
                        return NifflerResponse(
                            status=ResponseStatus.LOGIN_FAILED,
                            content={"message": "登陆失败"},
                            raw=res,
                        )
                    case 429:  # 触发频率限制
                        return NifflerResponse(
                            status=ResponseStatus.TRY_LATER,
                            content={"message": "触发频率限制"},
                            raw=res,
                        )
                    case 404 | 400:  # 请求异常
                        try:
                            message = res.json()
                        except Exception:
                            message = "请求异常"
                        return NifflerResponse(
                            status=ResponseStatus.BAD_REQUEST,
                            content={"message": message},
                            raw=res,
                        )
                    case success if 200 <= success <= 299:
                        pass
                    case error:  # 服务器异常
                        return NifflerResponse(
                            status=ResponseStatus.TRY_LATER,
                            content={"message": f"服务器异常({error})"},
                            raw=res,
                        )

                match method:
                    case HTTPMethod.GET | HTTPMethod.PATCH:
                        if res.content is None:
                            ntry += 1
                            await asyncio.sleep(10)
                            continue
                        else:
                            content = res.json()
                    case HTTPMethod.GET_LATER:
                        retry_after = float(res.headers.get("Retry-After", 0))
                        if retry_after:
                            await asyncio.sleep(retry_after + 10)
                            continue
                        content = res.json()
                    case _:
                        content = {}

                return NifflerResponse(
                    status=ResponseStatus.SUCCESS, content=content, raw=res
                )

            except Exception as e:
                ntry += 1
                self._logger.error(f"Error with {url} ({method}) {e}")
                await asyncio.sleep(10)

        _err = f"{method_core} {url} 超过最大重试次数, 请求失败"
        self._logger.error(_err, exc_info=True)
        raise ValueError(_err)

    async def get(
        self, url: str, params: dict | None = None, later: bool = False
    ) -> JSONLike:
        async with httpx.AsyncClient(
            cookies=httpx.Cookies({"t": (await self.get_user()).cookie}),
            base_url="<https://api.worldquantbrain.com>",
        ) as client:
            res = await self.send_request(
                client,
                HTTPMethod.GET_LATER if later else HTTPMethod.GET,
                url,
                params=params,
            )

        match res.status:
            case ResponseStatus.SUCCESS:
                return res.content
            case ResponseStatus.BAD_REQUEST:
                raise BadRequestError(res.content)
            case ResponseStatus.LOGIN_FAILED:
                raise TokenExpiredError("登陆已过期")
            case ResponseStatus.TRY_LATER:
                await asyncio.sleep(10 * 12)
                return await self.get(url, params)
            case _:
                return res.content

    async def batch_update_or_create(
        self, instances: list[T], model: type[T], fields: list[str] | None = None
    ) -> list[T]:
        """批量创建或更新外部模型

        外部模型具有唯一性, 包含唯一外部 ID, 该方法实现了外部模型的批量落表.

        Parameters
        ----------
        instances : list[T] 更新对象, 必须包含 id 否则无法更新
        model : type[T] 目标模型 (数据库表)
        fields : list[str] 要更新的字段 (默认为除了 id 和 created_at 之外的全部字段)
        """

        if fields is None:
            fields = [
                p
                for p in model._meta.fields_map.keys()
                if p not in ["id", "created_at"]
            ]

        ids_all = [i.id for i in instances if i.id is not None]
        ids_update = await model.filter(id__in=ids_all).values_list("id", flat=True)
        ids_create = [i for i in ids_all if i not in ids_update]

        to_update = [ins for ins in instances if ins.id in ids_update]
        to_create = [ins for ins in instances if ins.id in ids_create]

        if to_update:
            await model.bulk_update(to_update, fields=fields, batch_size=50)

        if to_create:
            await model.bulk_create(to_create, batch_size=50)

        return await model.filter(id__in=ids_all).all()

    async def setup(self) -> Self:
        try:
            await Tortoise.init(
                config={
                    "connections": {"default": self._connection},
                    "apps": {
                        "niffler": {
                            "models": ["models"],
                            "default_connection": "default",
                        }
                    },
                    "use_tz": True,
                    "timezone": "Asia/Shanghai",
                },
                _enable_global_fallback=True,
            )
        except ConfigurationError:
            self._logger.debug("Tortoise 已完成初始化")
        await Tortoise.generate_schemas(safe=True)
        self._logger.info(f"本地因子库已连接: {self._connection}")
        self._user = await self.get_user()
        return self

    async def close(self) -> None:
        await Tortoise.close_connections()

    async def get_user(self, force: bool = False) -> User:
        if (
            (self._user is not None)  # 已完成初始化
            and (self._user.cookie is not None)  # Cookie 不为空
            and (
                datetime.now(ZoneInfo("Asia/Shanghai")) - self._user.updated_at
                < timedelta(seconds=self._user.expiry) - self._expire_redun
            )  # Cookie 未过期
            and not force
        ):
            self._logger.debug("获取权鉴缓存")
            return self._user
        else:
            user = await User.get_or_none(username=self._username)
            if (
                (user is not None)
                and (user.cookie is not None)
                and (
                    datetime.now(ZoneInfo("Asia/Shanghai")) - user.updated_at
                    < timedelta(seconds=user.expiry) - self._expire_redun
                )
            ):
                self._logger.debug("读取本地权鉴")
                self._user = user
                return user
            else:
                self._logger.debug("权鉴已过期, 重新登录")
                return await self.login()

    async def login(self) -> User:
        """登录"""
        async with httpx.AsyncClient(
            auth=httpx.BasicAuth(self._username, self._password),
            base_url="<https://api.worldquantbrain.com>",
        ) as client:
            response = await self.send_request(
                client, HTTPMethod.POST, "/authentication"
            )
            if response.status != ResponseStatus.SUCCESS or not isinstance(
                response.content, dict
            ):
                raise ValueError("登陆失败, 请检查用户名和密码是否正确")

        self._user, _ = await User.update_or_create(
            id=response.raw.json()["user"]["id"],
            defaults=dict(
                username=self._username,
                password=self._password,
                permissions=response.raw.json().get("permissions"),
                cookie=response.raw.cookies.get("t"),
                expiry=response.raw.json()["token"]["expiry"],
            ),
        )
        self._logger.debug(f"登陆成功: {response.raw.json()['user']['id']}")
        return self._user

    async def get_alpha(
        self, alpha_id: str, force: bool = False
    ) -> RegularAlpha | SuperAlpha:
        if not force:
            ra = await RegularAlpha.get_or_none(id=alpha_id)
            sa = await SuperAlpha.get_or_none(id=alpha_id)
            if ra is not None:
                self._logger.debug(f"读取本地 RA: {ra}")
                return ra
            if sa is not None:
                self._logger.debug(f"读取本地 SA: {sa}")
                return sa
        rcd = await self.get(url=f"/alphas/{alpha_id}")
        assert isinstance(rcd, dict), "获取因子失败"
        if rcd["type"] == "SUPER":
            alpha = SuperAlpha.from_record(rcd)
            keys = SuperAlpha.get_keys_from_record(rcd)
            alpha_batch = await self.batch_update_or_create(
                [alpha], SuperAlpha, [k for k in keys if k not in ["id"]]
            )
        elif rcd["type"] == "REGULAR":
            alpha = RegularAlpha.from_record(rcd)
            keys = RegularAlpha.get_keys_from_record(rcd)
            alpha_batch = await self.batch_update_or_create(
                [alpha], RegularAlpha, [k for k in keys if k not in ["id"]]
            )
        else:
            raise ValueError(f"alpha type {rcd['type']} not supported")
        alpha_get = alpha_batch[0]
        self._logger.debug(f"在线加载因子: {alpha_get}")
        return alpha_get

```

## 代码解释

本人自认为码力还可以，本系列教程代码全为古法编程，匠心汇聚，😆。相信熟悉Python的同学纯看代码应该就能看懂了，重点位置已经放了注释，这里简单讲解一下比较核心的内容：

### 高级HTTP请求接口： `*send_request*`

该方法利用HTTPX库，根据Brain平台的特点，包装了一个高级HTTP请求接口，支持 Get/Get-Later/Post/Patch, 实现了自动重试和超限休眠，尤其是Get-Later，用来请求需要长时等待的资源，例如PNL曲线。新顾问应该会注意到，GetPnl时第一次会报错返回空值，需要再Get一次才可以，遇到网络波动的时候甚至需要重复GetN次，不过现在这些问题都被这一个接口解决了。

### 因子库更新接口： `*batch_update_or_create*`

批量创建或更新外部模型，外部模型具有唯一性, 包含唯一外部 ID, 该方法实现了外部模型的批量落表。

我使用过很多种ORM，很奇怪的是大家都缺少这样一个接口，都需要人工造轮子，主要实现的功能是：批量更新某张表（例如RegularAlpha表），如果某条记录存在，则用数据字段更新，如果不存在，则新增。如果一个一个更新，受限于数据库接口设计，会贼慢，该方法主要包装了数据库ORM的批量更新能力。

### 两个对外接口： `get_user`  和  `get_alpha`

这两个接口是我们日常挖因子真正要用的接口，一个是获取当前用户，一个是获取某个因子。

获取因子没什么好讲的，大家看代码即可，获取用户这里我解决了Cookie 持久化的痛点，大概的逻辑是：

1. 当前内存中已有user，且Cookie未过期：直接获取User
2. 库里有Cookie且未过期（Cookie的过期时间会从登陆接口返回来，一般是3.5小时，我们只需要记住登陆的时间，然后加上过期时间就可以知道现在是否过期，我这里设计了半小时的冗余，防止时间误差导致的错误）：直接读取Cookie，获取User
3. 库里没有Cookie或者Cookie已过期：重新登陆，更新库里的Cookie，获取User

此外，为了防止异步时不同协程的冲突，在发送API请求时遇到 401, 429 错误时，也会首先尝试重新获取 User，防止其他协程已经更新过 Cookie，然后本协程又要更新一次的情况。（不过其实这些大家都不用管，我已经都处理好了）

## 下篇预告

篇幅有限，本篇教程就更新到这里，其实目前核心内容已经更新的差不多了，大家可以根据自己的需要进一步完善本地因子库，主要需要完善的内容是 client 部分，例如我们还需要实现：

- `get_all` : 一次性获取分页数据，例如dataset、alphaset、operators等都是分页的，我们需要一个接口来一次性获取分页内容
- `get_alphaset` : 获取因子集，众所周知，一次检索条件的因子最多10000个，为了获取全部因子，我们需要给这个接口增加按日期自动分片的能力，并且由于获取因子其实还挺慢的，我们还需要利用因子库，跳过哪些已经获取过的因子
- `get_pnl`   `get_perf`   `get_stats`  等因子指标
- `get_self_corr`   `get_prod_corr`  获取因子相关性
- `update_alpha`  更新因子信息，例如标签批量修改，因子批量打标（我喜欢把那些OS表现差的因子标为红色，在组Own-SA时用条件过滤掉），OsmoisPoint 打分、增加因子描述等，大家可以看数据库中预留的字段，都是可以修改的
- 还有最重要的： `simulate`  因子回测接口

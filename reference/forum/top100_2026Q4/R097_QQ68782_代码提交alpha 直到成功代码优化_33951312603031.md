## 帖子正文

最近提交alpha很慢, 一直在转圈圈, 分享一下我的代码提交方法, 用了很长一段时间, 基本上只要能check通过, 就能提交成功;

**代码**

```
def submit_alpha(alpha_id):    '''提交alpha, 一直重试到成功;     '''    s = login()    TOTTAL_RETRY_COUNT = 100000    retry_count = 0    while retry_count < TOTTAL_RETRY_COUNT:        try:            url = "https://api.worldquantbrain.com/alphas/" + alpha_id + "/submit"            logger.debug(f'url: {url}')            res = s.post(url)            if res.status_code == 400 and "The plain HTTP request was sent to HTTPS port" in res.text:                # 已经提交过了                logger.info("提交中...")                res.headers["Retry-After"] = 1.0            logger.debug("enter retry-after")            while True:                if "retry-after" in res.headers:                    sleep_time = max(float(res.headers["Retry-After"]), 3)                    time.sleep(sleep_time)                    print(".", end="", flush=True)                    res = s.get(url)                else:                    logger.info("break")                    break            logger.info(f"submit alpha {alpha_id} {res} {res.text}")            if res.status_code == 429:                logger.info(f"sleep")                sleep(60)                retry_count += 1                continue            if res.status_code == 401:                s = login()                retry_count += 1                continue            if res.status_code == 404:                # 超时                retry_count += 1                continue            if res.status_code // 100 == 5:                # 网关错误                retry_count += 1                sleep(5)                continue            if res.status_code == 403:                # fail                logger.info(f"{alpha_id} submit fail")                fail_checks = []                try:                    checks = res.json()["is"]["checks"]                    fail_checks = list(                        filter(lambda x: x["result"] in ["FAIL"], checks))                except Exception as e:                    logger.info(f'fail_checks exception={e}')                logger.info(f"fail_checks={fail_checks}")                # fail_checks=[{'name': 'REGULAR_SUBMISSION', 'result': 'FAIL', 'limit': 4, 'value': 4}]                if len(list(filter(lambda x: x["name"] in ["REGULAR_SUBMISSION", "SUPER_SUBMISSION"], fail_checks))) > 0:                    logger.info("SUBMISSION 超过限制 "+fail_checks)                    return                # set_check_fail_alpha(alpha_id, fail_checks, res.json())                return            if res.status_code == 200:                logger.info(f"{alpha_id} submit success")                return res.status_code            if res.status_code//100 != 2:                logger.info(f"submit alpha not 200  {alpha_id} {res}")                return            return        except requests.exceptions.RequestException as e:            logger.error(f"submit alpha RequestException {                alpha_id} retry_count={retry_count} {e}")            retry_count += 1            sleep(10)            continue        except Exception as e:            logger.error(f"submit alpha Exception {alpha_id} {e}")            return
```

这个脚本主要用于当天“断粮”、没有提交满配额时做保底补充。前面即使已经做过 PC 过滤，手里仍可能剩下一些 PC 高于 0.7、但 Sharpe 仍然比较高的候选 Alpha；这种 Alpha 不一定适合作为主力提交，但在当天配额没满时，也可以按自己设定的指标条件再筛一遍，作为补充提交。

设置好 Region、Delay、Universe、指标阈值和本次提交数量后，脚本会自动：

- 拉取符合条件的候选 Alpha；
- 自动跳过 IS FAIL；

- 支持按 Sharpe 从高到低，或从低到高排序；
- 自动依次提交 Alpha；
- 持续轮询最终提交状态；
- 提交成功后，VS Code 自动语音播报 Alpha ID；
- 自动记录失败 Alpha，避免后面重复提交。

def safe_nested_get(d, path, default=None):

cur = d

for key in path.split("."):

if not isinstance(cur, dict) or key not in cur:

return default

cur = cur[key]

return cur

def compare_value(actual, op, target):

if actual is None:

return False

try:

actual = float(actual)

target = float(target)

except Exception:

return False

if op == ">":

return actual > target

elif op == ">=":

return actual >= target

elif op == "<":

return actual < target

elif op == "<=":

return actual <= target

elif op in ("=", "=="):

return actual == target

else:

raise ValueError(f"不支持的比较符: {op}")

def value_in_allowed(actual, allowed, tol=1e-12):

if allowed is None:

return True

if actual is None:

return False

for x in allowed:

try:

if abs(float(actual) - float(x)) <= tol:

return True

except Exception:

if actual == x:

return True

return False

def alpha_passes_filters(

alpha,

metric_filters=None,

allowed_decay=None,

allowed_truncation=None,

):

"""

metric_filters 示例:

{

"sharpe":   (">", 1.2),

"fitness":  (">", 1.0),

"margin":   (">", 0.0002),

"drawdown": ("<", 0.15),

"turnover": ("<", 0.45),

}

"""

if metric_filters is None:

metric_filters = {}

field_map = {

"sharpe": "is.sharpe",

"fitness": "is.fitness",

"margin": "is.margin",

"drawdown": "is.drawdown",

"turnover": "is.turnover",

}

for metric_name, rule in metric_filters.items():

if metric_name not in field_map:

print(f"[WARN] 未知指标: {metric_name}，该 alpha 直接跳过")

return False

if not isinstance(rule, (tuple, list)) or len(rule) != 2:

print(f"[WARN] {metric_name} 的规则格式错误，应为 ('>', 1.2)")

return False

op, target = rule

actual = safe_nested_get(alpha, field_map[metric_name], None)

if actual is None:

print(f"[WARN] alpha {alpha.get('id')} 缺少字段 {field_map[metric_name]}")

return False

if not compare_value(actual, op, target):

return False

decay = safe_nested_get(alpha, "settings.decay", None)

if not value_in_allowed(decay, allowed_decay):

return False

truncation = safe_nested_get(alpha, "settings.truncation", None)

if not value_in_allowed(truncation, allowed_truncation):

return False

return True

def calc_returns_from_fitness(sharpe, fitness, turnover):

"""

WQB 公式：

Fitness = Sharpe * sqrt(abs(Returns) / max(Turnover, 0.125))

反推：

abs(Returns) = (Fitness / Sharpe)^2 * max(Turnover, 0.125)

对当前提交筛选用途，默认 sharpe、fitness 为正时，将 returns 视为正收益率。

返回小数形式：

0.02 = 2%

0.12 = 12%

"""

if sharpe is None or fitness is None or turnover is None:

return None

try:

sharpe = float(sharpe)

fitness = float(fitness)

turnover = float(turnover)

except Exception:

return None

if sharpe == 0:

return None

ret = (fitness / sharpe) ** 2 * max(turnover, 0.125)

return round(float(ret), 6)

def build_alpha_metrics(alpha_detail):

sharpe = safe_nested_get(alpha_detail, "is.sharpe", None)

fitness = safe_nested_get(alpha_detail, "is.fitness", None)

turnover = safe_nested_get(alpha_detail, "is.turnover", None)

drawdown = safe_nested_get(alpha_detail, "is.drawdown", None)

margin = safe_nested_get(alpha_detail, "is.margin", None)

returns = calc_returns_from_fitness(

sharpe=sharpe,

)

return {

"sharpe": sharpe,

"fitness": fitness,

"turnover": turnover,

"returns": returns,

"drawdown": drawdown,

"margin": margin,

}

def passes_submit_filter(

metrics,

min_margin=0.0015,

min_returns_turnover_ratio=0.08,

):

"""

提交条件：

1. margin > min_margin

2. returns >= turnover * min_returns_turnover_ratio

例子：

turnover = 0.25

min_returns_turnover_ratio = 0.08

最低 returns = 0.25 * 0.08 = 0.02，即 2%

"""

reasons = []

try:

returns = float(metrics.get("returns"))

except Exception:

reasons.append("returns 缺失或无法转 float")

return False, reasons

try:

turnover = float(metrics.get("turnover"))

except Exception:

reasons.append("turnover 缺失或无法转 float")

return False, reasons

try:

margin = float(metrics.get("margin"))

except Exception:

reasons.append("margin 缺失或无法转 float")

return False, reasons

ratio = float(min_returns_turnover_ratio)

if ratio > 1:

ratio = ratio / 100.0

min_required_returns = turnover * ratio

if returns < min_required_returns:

reasons.append(

f"returns={returns} < turnover*ratio={min_required_returns:.6f} "

f"(turnover={turnover}, ratio={ratio:.2%})"

)

if margin <= min_margin:

reasons.append(

f"margin={margin} <= min_margin={min_margin}"

)

return len(reasons) == 0, reasons

def submit_alpha(s, alpha_id, max_retry=20):

"""

提交 alpha。

注意：

- HTTP 200/201/202 只能说明 submit 请求被接受

- 不能说明最终提交成功

- 最终成功/失败必须靠后续 GET submit_url 的结果判断

"""

global GLOBAL_SESSION

submit_url = f" [https://api.worldquantbrain.com/alphas/{alpha_id}/submit](https://api.worldquantbrain.com/alphas/{alpha_id}/submit) "

for attempt in range(1, max_retry + 1):

s = GLOBAL_SESSION or s

try:

r = s.post(submit_url, timeout=30)

retry = r.headers.get("Retry-After")

if retry is not None:

wait = float(retry)

print(f"[SUBMIT] {alpha_id} Retry-After={wait} 秒，等待后 GET submit 状态")

time.sleep(wait)

r = s.get(submit_url, timeout=30)

if r.status_code in (200, 201, 202):

text = r.text or ""

print(f"[SUBMIT] {alpha_id} 提交请求已接受，status={r.status_code}")

try:

js = r.json()

if isinstance(js, dict):

js["_raw_status_code"] = r.status_code

js["_headers"] = dict(r.headers)

js["_raw_text"] = text[:2000]

js["alpha_id"] = alpha_id

js["submit_url"] = submit_url

js["status_code"] = r.status_code

return js

except Exception:

pass

return {

"alpha_id": alpha_id,

def check_submit_status(alpha_id, submit_info=None, debug=False):

"""

def parse_submit_status(result):

"""

解析 submit endpoint 返回结果。

规则：

- HTTP 403 + checks 中有 FAIL => FAIL

- checks 中有 FAIL => FAIL

- checks 存在且全部 PASS/WARNING，无 FAIL => SUCCESS

- Retry-After / 202 / 204 / pending 类状态 => PENDING

"""

if result is None:

return "UNKNOWN"

if not isinstance(result, dict):

return "PENDING"

raw_status_code = result.get("_raw_status_code", result.get("status_code"))

if result.get("_pending_by_retry_after"):

return "PENDING"

status = (

result.get("status")

or result.get("state")

or result.get("result")

or result.get("submitStatus")

)

if status is not None:

s = str(status).upper()

if s in ("SUCCESS", "SUBMITTED", "PASS", "DONE", "COMPLETE", "COMPLETED"):

return "SUCCESS"

if s in ("FAIL", "FAILED", "ERROR", "REJECTED"):

return "FAIL"

if s in ("PENDING", "QUEUE", "QUEUED", "PROCESSING", "RUNNING", "IN_PROGRESS"):

return "PENDING"

checks = (

result.get("checks")

or result.get("submitChecks")

or result.get("submissionChecks")

or safe_nested_get(result, "is.checks", [])

or []

)

if isinstance(checks, dict):

checks = list(checks.values())

if isinstance(checks, list) and checks:

has_fail = False

has_pending = False

has_pass_or_warning = False

for chk in checks:

if not isinstance(chk, dict):

continue

chk_result = (

chk.get("result")

or chk.get("status")

or chk.get("state")

)

chk_s = str(chk_result).upper()

if chk_s in ("FAIL", "FAILED", "ERROR", "REJECTED"):

has_fail = True

elif chk_s in ("PENDING", "QUEUE", "QUEUED", "PROCESSING", "RUNNING", "IN_PROGRESS"):

has_pending = True

elif chk_s in ("PASS", "PASSED", "SUCCESS", "DONE", "COMPLETE", "COMPLETED", "WARNING"):

has_pass_or_warning = True

if has_fail:

return "FAIL"

if has_pending:

return "PENDING"

if has_pass_or_warning:

return "SUCCESS"

if raw_status_code == 403:

return "FAIL"

if raw_status_code in (201, 202, 204):

return "PENDING"

raw_text = str(result.get("_raw_text", "")).upper()

if raw_text:

if "FAIL" in raw_text or "FAILED" in raw_text or "REJECTED" in raw_text:

return "FAIL"

if "PENDING" in raw_text or "QUEUE" in raw_text or "PROCESSING" in raw_text:

return "PENDING"

if "SUCCESS" in raw_text or "SUBMITTED" in raw_text or "COMPLETE" in raw_text:

return "SUCCESS"

return "PENDING"

def poll_submit_result(alpha_id, submit_info=None, max_wait_seconds=1800, debug_first=True):

"""

轮询 submit 状态：

- 前 1 分钟：每 10 秒查一次

- 超过 1 分钟后：1分钟、2分钟、3分钟递增等待

- 最多 30 分钟

"""

start = time.time()

step_minute = 1

poll_count = 0

while True:

elapsed = time.time() - start

if elapsed >= max_wait_seconds:

print(f"[POLL] {alpha_id} 超过 30 分钟没有明确状态")

return "TIMEOUT", None

if elapsed < 60:

sleep_sec = 10

else:

sleep_sec = min(step_minute * 60, 30 * 60)

step_minute += 1

print(f"[POLL] {alpha_id} 等待 {sleep_sec} 秒后查询 submit 状态")

time.sleep(sleep_sec)

poll_count += 1

result = check_submit_status(

alpha_id,

submit_info=submit_info,

debug=(debug_first and poll_count == 1),

)

if result is None:

print(f"[POLL] {alpha_id} 没有状态返回，elapsed={elapsed:.0f}s")

continue

status = parse_submit_status(result)

if status == "SUCCESS":

return "SUCCESS", result

if status == "FAIL":

return "FAIL", result

print(f"[POLL] {alpha_id} 当前状态={status}，继续等待")

def extract_submit_fail_reasons(result):

if not isinstance(result, dict):

return ["UNKNOWN_FAIL"]

reasons = []

checks = (

result.get("checks")

or result.get("submitChecks")

or result.get("submissionChecks")

or safe_nested_get(result, "is.checks", [])

or []

)

if isinstance(checks, dict):

checks = list(checks.values())

if isinstance(checks, list) and checks:

for chk in checks:

if not isinstance(chk, dict):

continue

chk_result = (

chk.get("result")

or chk.get("status")

or chk.get("state")

)

if str(chk_result).upper() == "FAIL":

name = chk.get("name", "UNKNOWN_CHECK")

limit = chk.get("limit", None)

value = chk.get("value", None)

message = chk.get("message") or chk.get("description") or ""

parts = [name]

if value is not None:

parts.append(f"value={value}")

if limit is not None:

parts.append(f"limit={limit}")

if message:

parts.append(str(message))

reasons.append(" | ".join(parts))

if not reasons:

message = (

result.get("message")

or result.get("error")

or result.get("detail")

or result.get("_raw_text")

)

if message:

reasons.append(str(message)[:500])

if not reasons:

reasons.append("UNKNOWN_FAIL")

return reasons

## 使用方法

1. 将 Notebook 和  `check_username.txt`  放在同一个运行目录下。
2. 在  `check_username.txt`  中填写自己的 WQB 账号和密码，格式如下：

```
{
  "username": "你的邮箱",
  "password": "你的密码"
}

```

1. 打开 Notebook，先按自己的需求修改最后运行区的参数，包括 Region、Delay、Universe、Neutralization、Sharpe / Fitness / Margin / Turnover / Drawdown 阈值，以及本次最多提交数量。
2. 运行即可。脚本会自动登录、拉取候选 Alpha、跳过 IS FAIL、按指标筛选，并按设定顺序自动提交。
3. 支持按 Sharpe 从高到低或从低到高提交；提交后会继续轮询最终状态。成功时 VS Code 会语音播报对应 Alpha ID，失败结果也会自动记录。

适合当天没有提交满配额时使用：包括 PC 过滤后剩下的候选，或者部分 PC 高于 0.7、但 Sharpe 仍然较高，想作为补充再筛一轮提交的 Alpha。

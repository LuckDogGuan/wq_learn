本文摘要
在利用 AI Agent（如 Antigravity / Claude / Cursor）挖掘 WorldQuant Brain (WQB) 因子时，LLM 常常面临
官方规则记不住、冷门算子乱编、报错信息不会解、社区前人踩坑经验无法实时召回
等痛点。
本项目
wqb_mcp_project
搭建了一整套
端到端的 WQB 量化知识库与 MCP 检索系统
：
全自动化数据清洗与生产流水线
：涵盖从论坛长文自动爬取、点赞/时间热度过滤，到大模型痛点驱动的原子化 QA 提取；
语法敏感的切块与离线混合向量库
：采用代码块防截断的 Markdown 智能切分，结合
bge-small-zh-v1.5
与
LanceDB
实现低延迟 Hybrid Search；
极速唤醒的 MCP 知识服务
：针对 IDE 经常出现的 MCP 初始化握手超时问题，实现极致懒加载，向上层 Agent 提供官方指令级规则与社区干货长文的原生无缝召回。
痛点与背景
大家在用 AI 辅助手搓 Alpha 或自动优化表达式时，基本都踩过这些坑：
官方约束繁杂且动态更新
：比如各个地区的 Turnover、Sharpe、Coverage、Sub-universe 门槛经常变动，LLM 仅凭原生参数库极易给出通不过检查的废因子。
社区智慧分散沉底
：中文论坛有大量神仙帖子（如关于 Robust Universe 优化、低 PC 因子降相关、特定算子在缺失值场景下的填补技巧），但往往淹没在几千篇帖子里，人工去搜效率极低。
传统 RAG 接入 MCP 频频超时断连
：在 MCP 进程启动时如果直接同步
import torch
、加载几个 G 的向量模型，IDE 客户端往往在 10 秒握手超时内直接判定 MCP 崩溃退出。
为了彻底解决“知识精准供给”的问题，我们设计并落地了本项目。

三层：
核心模块剖析
1. 论坛数据处理流水线（forum_post/）
一套高质量的 RAG 核心在于“垃圾进，垃圾出；干货进，黄金出”。流水线分为清晰的三步：
抓取端
forum_post/step1_download_forum.py
：利用 Playwright 模拟登录 Brain 平台，增量同步社区帖子并转化为 Markdown，文件名规整记录发帖时间、点赞数、评论数与唯一 Post ID。
过滤端
forum_post/step2_filter_low.py
：设置动态过滤阈值。对近 1 个月新帖降低门槛（只要点赞或评论其一 ≥ 5 即保留），对 1 个月前的历史帖子提高硬性指标（双向过滤低于 10 的水帖），输出到
filter_post/
。
LLM 语义提取端
forum_post/step3_summary.py
：调用 Gemini Flash 对清洗后的长文做原子化 QA 重构。
痛点场景驱动
：严禁模糊提问，将真实报错（如 Turnover 超限、数据缺失）直接设为 Question。
关键词黑话补全
：自动附带平台黑话（SA、RA、正交化、过拟合），大幅提升下游 BM25 词法命中率。
2. 智能切块与多路向量同步（sync_db.py）
在入库脚本
sync_db.py
中，重点解决了传统切块工具硬切代码块的问题：
代码保护型 Markdown 切块算法

② 官方权威约束检索 search_wqb_official_guidelines

import os
import random
import datetime
import traceback
from mcp.server.fastmcp import FastMCP
mcp = FastMCP("WQB-Knowledge-Base")
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
# ==========================================
# 极致懒加载器：只有在工具被调用时，才导入重型库并连接
# ==========================================
_model_instance = None
_db_tables = None
def get_tables():
"""延迟加载 LanceDB 数据库"""
global _db_tables
if _db_tables is None:
import lancedb
DB_PATH = os.path.join(BASE_DIR, "lancedb_data")
db = lancedb.connect(DB_PATH)
_db_tables = {
"official": db.open_table("wqb_official_guides"),
"forum": db.open_table("wqb_forum_experiences")
}
return _db_tables
def get_embedding_model():
"""延迟加载 PyTorch 和 BGE 向量模型"""
global _model_instance
if _model_instance is None:
import warnings
import logging
os.environ["TOKENIZERS_PARALLELISM"] = "false"
os.environ["HF_HUB_DISABLE_PROGRESS_BARS"] = "1"
os.environ["HF_HUB_OFFLINE"] = "1"
warnings.filterwarnings("ignore")
logging.getLogger("httpx").setLevel(logging.ERROR)
from sentence_transformers import SentenceTransformer
_model_instance = SentenceTransformer("BAAI/bge-small-zh-v1.5")
return _model_instance
# ==========================================
# 工具定义
# ==========================================
@mcp.tool()
def search_wqb_official_guidelines(query: str, top_k: int = 15) -> str:
"""【绝对最高优先级调用】检索 WorldQuant Brain (WQB) 官方标准的规则约束与动作指令"""
log_file = os.path.join(BASE_DIR, "mcp_trigger_debug.log")
try:
with open(log_file, "a", encoding="utf-8") as f:
f.write(f"[{datetime.datetime.now()}] 💡 search_wqb_official 触发 | Query: {query}\n")
except Exception: pass
model = get_embedding_model()
tables = get_tables()
table_official = tables["official"]
instruction = "为这个句子生成表示以用于检索相关文章："
query_vector = model.encode(instruction + query).tolist()
try:
results = (
table_official.search(query_type="hybrid")
.text(query)
.vector(query_vector)
.limit(top_k)
.to_list()
)
except Exception as e:
results = table_official.search(query_vector).limit(top_k).to_list()
if not results: return "⚠️ 未找到相关规则。"
response = "💡 【官方指令级知识库】检索结果：\n"
response += f"🔔 [系统强制指令]: 今天是 {datetime.date.today()}。请优先采纳最新记录。\n\n"
for idx, row in enumerate(results):
response += f"【官方规则 {idx + 1}】\n{row['text']}\n" + "=" * 40 + "\n"
return response
@mcp.tool()
def search_wqb_forum_experience(query: str) -> str:
top_k = 10
"""【次高优先级 / 辅助补充调用】检索 WorldQuant Brain (WQB) 社区论坛经验"""
import re
import glob
model = get_embedding_model()
tables = get_tables()
table_forum = tables["forum"]
instruction = "为这个句子生成表示以用于检索相关文章："
query_vector = model.encode(instruction + query).tolist()
# 设定候选文章池为 30 篇（如果 top_k 被修改，动态取 max(top_k * 3, 30)）
target_pool_size = max(top_k * 3, 30)
current_limit = max(target_pool_size * 10, 300)
max_limit = 5000
unique_article_data = []
while len(unique_article_data) < target_pool_size and current_limit <= max_limit:
try:
results_pool = (
table_forum.search(query_type="hybrid")
.text(query)
.vector(query_vector)
.limit(current_limit)
.to_list()
)
except Exception as e:
results_pool = table_forum.search(query_vector).limit(current_limit).to_list()
if not results_pool:
break
unique_article_data = []
seen_ids = set()
for row in results_pool:
text = row.get('text', '')
match = re.search(r'posts/(\d+)', text)
if match:
article_id = match.group(1)
if article_id not in seen_ids:
seen_ids.add(article_id)
header_parts = text.split('\nQ:')
header = header_parts[0].strip()
unique_article_data.append((article_id, header))
# 捞够候选池数量就停止
if len(unique_article_data) >= target_pool_size:
break
if len(unique_article_data) >= target_pool_size:
break
if len(results_pool) < current_limit:
break
current_limit *= 2
if not unique_article_data:
return "未能从检索结果中提取到有效的文章ID链接。"
# 应用随机抽样策略：保留最相关的前7篇，剩下的从其余文章中随机抽取
if len(unique_article_data) > top_k:
core_count = min(7, len(unique_article_data))
core_results = unique_article_data[:core_count]  # 保留分数最高的 7 篇
rest_pool = unique_article_data[core_count:]     # 剩下的池子 (比如 30 - 7 = 23 篇)
# 还需要抽取的数量（确保总数等于 top_k）
need_sample = top_k - len(core_results)
sampled_rest = random.sample(rest_pool, min(need_sample, len(rest_pool)))
final_article_data = core_results + sampled_rest
else:
final_article_data = unique_article_data
response = f"💬 【社区高纯度经验长文原文】检索结果（共关联 {len(final_article_data)} 篇文章）：\n\n"
md_dir = os.path.join(BASE_DIR, "forum_post", "filter_post")
for idx, (article_id, header) in enumerate(final_article_data):
# 区分是核心命中还是随机扩充（在头部标记一下）
is_core = "核心命中" if idx < 7 else "随机扩充"
response += f"==================== 【社区经验文档 {idx + 1} ({is_core}) | 文章ID: {article_id}】 ====================\n"
# 写入元数据头部
response += header + "\n\n"
file_pattern = os.path.join(md_dir, f"*{article_id}*.md")
matched_files = glob.glob(file_pattern)
if matched_files:
try:
with open(matched_files[0], 'r', encoding='utf-8') as f:
content = f.read()
response += "【文章正文内容】\n"
response += content + "\n\n"
except Exception as e:
response += f"[⚠️ 读取文件失败: {str(e)}]\n\n"
else:
response += f"[⚠️ 未在本地知识库中找到该ID对应的md文件: {article_id}]\n\n"
return response
if __name__ == "__main__":
mcp.run(transport="stdio")
